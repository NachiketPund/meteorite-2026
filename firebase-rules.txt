// Firestore rules
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function isAdmin() {
      return request.auth != null &&
        exists(/databases/$(database)/documents/admins/$(request.auth.uid));
    }

    match /admins/{uid} {
      allow read: if isAdmin();
      allow write: if false;
    }

    match /registrations/{id} {
      // Anonymous/authenticated participant creates only their own pending registration.
      allow create: if request.auth != null
        && request.resource.data.uid == request.auth.uid
        && request.resource.data.status == "pending";

      // Participant cannot read other registrations.
      allow read: if isAdmin();

      // Only admin can update verification status.
      allow update: if isAdmin();

      allow delete: if isAdmin();
    }
  }
}

// Storage rules
rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {
    match /paymentProof/{uid}/{fileName} {
      allow write: if request.auth != null
        && request.auth.uid == uid
        && request.resource.size < 5 * 1024 * 1024
        && request.resource.contentType.matches('image/.*');

      allow read: if request.auth != null &&
        exists(/databases/(default)/documents/admins/$(request.auth.uid));
    }
  }
}
